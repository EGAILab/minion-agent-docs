  const fs = require("fs/promises");
  const nodePath = require("path");

  async function lsRepro(dirPath, effectiveLimit) {
    const entries = await fs.readdir(dirPath);
    entries.sort((a, b) => a.toLowerCase().localeCompare(b.toLowerCase()));
    const results = [];
    let entryLimitReached = false;
    for (const entry of entries) {
      if (results.length >= effectiveLimit) {
        entryLimitReached = true;
        break;
      }
      const fullPath = nodePath.join(dirPath, entry);
      let suffix = "";
      try {
        const entryStat = await fs.stat(fullPath);
        if (entryStat.isDirectory()) suffix = "/";
      } catch {
        continue;
      }
      results.push(entry + suffix);
    }
    return { results, entryLimitReached };
  }

(filesystem.py:368-461):

  import os, stat as _stat

  class _UnsupportedFileType(Exception):
      pass

  def _file_kind_from_stat(st):
      if _stat.S_ISREG(st.st_mode): return "FILE"
      if _stat.S_ISDIR(st.st_mode): return "DIRECTORY"
      if _stat.S_ISLNK(st.st_mode): return "SYMLINK"
      return None

  def _file_info_sync(path):
      st = os.lstat(path)
      kind = _file_kind_from_stat(st)
      if kind is None:
          raise _UnsupportedFileType
      return {"name": os.path.basename(path), "path": path, "kind": kind}

  def _list_dir_sync(path):
      infos = []
      with os.scandir(path) as entries:
          for entry in entries:
              try:
                  infos.append(_file_info_sync(entry.path))
              except _UnsupportedFileType:
                  continue
      return infos
